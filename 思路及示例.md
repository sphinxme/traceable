# Yjs + Svelte 5 响应式封装设计

## 核心思路

1. **`createSubscriber`** 用于订阅 yjs 的变更事件，让 Svelte 知道何时重新读取
2. **getter** 在读取时触发 Svelte 的依赖追踪，同时从 yMap 中取最新值
3. **setter** 直接写入 yMap，yjs 的事件机制会通知订阅者
4. **懒代理模式** 对 children/events 等关联字段返回可迭代的响应式代理对象

## 实现

### 基础工具：响应式 YMap 订阅器

```typescript
// reactive-ymap.ts
import { createSubscriber } from 'svelte/reactivity';
import * as Y from 'yjs';

/**
 * 为一个 Y.Map 创建一个 subscribe 函数，
 * 当 Y.Map 发生任何变化时通知 Svelte 重新读取。
 * 
 * createSubscriber 返回的 subscribe 函数：
 *   - 在 Svelte effect/template 中被调用时，注册依赖追踪
 *   - 当 update 被调用时，通知 Svelte 依赖已变化
 */
export function createYMapSubscriber(yMap: Y.Map<any>): () => void {
  const subscribe = createSubscriber((update) => {
    const handler = () => update();
    yMap.observe(handler);
    return () => yMap.unobserve(handler);
  });
  return subscribe;
}

/**
 * 为一个 Y.Array 创建订阅器
 */
export function createYArraySubscriber(yArray: Y.Array<any>): () => void {
  const subscribe = createSubscriber((update) => {
    const handler = () => update();
    yArray.observe(handler);
    return () => yArray.unobserve(handler);
  });
  return subscribe;
}

/**
 * 为一个顶层 Y.Map（如 ydoc.getMap('tasks')）创建订阅器，
 * 用于监听 key 的增删（deep=false 的 observe 即可）。
 */
export function createYMapKeysSubscriber(yMap: Y.Map<any>): () => void {
  const subscribe = createSubscriber((update) => {
    const handler = () => update();
    yMap.observe(handler);
    return () => yMap.unobserve(handler);
  });
  return subscribe;
}
```

### Store：全局实体注册中心

```typescript
// store.ts
import * as Y from 'yjs';
import { Task } from './task.svelte';
import { Event } from './event.svelte';

/**
 * Store 是整个响应式封装的核心注册中心。
 * 
 * 它持有 ydoc 的引用以及三个顶层 Y.Map 的引用，
 * 并且维护 Task/Event 实例的缓存（同一个 id 始终返回同一个实例），
 * 避免重复创建和重复订阅。
 */
export class Store {
  readonly ydoc: Y.Doc;
  readonly tasksMap: Y.Map<Y.Map<any>>;
  readonly eventsMap: Y.Map<Y.Map<any>>;
  readonly textsMap: Y.Map<Y.Text>;

  // 实例缓存：保证同一个 id 返回同一个包装对象
  private taskCache = new Map<string, Task>();
  private eventCache = new Map<string, Event>();

  constructor(ydoc: Y.Doc) {
    this.ydoc = ydoc;
    this.tasksMap = ydoc.getMap('tasks') as Y.Map<Y.Map<any>>;
    this.eventsMap = ydoc.getMap('events') as Y.Map<Y.Map<any>>;
    this.textsMap = ydoc.getMap('texts') as Y.Map<Y.Text>;
  }

  /**
   * 根据 taskId 获取 Task 包装对象。
   * 使用缓存确保引用稳定性（对 Svelte 的 keyed each 友好）。
   */
  getTask(taskId: string): Task | undefined {
    const yMap = this.tasksMap.get(taskId);
    if (!yMap) return undefined;

    let task = this.taskCache.get(taskId);
    if (!task) {
      task = new Task(yMap, this);
      this.taskCache.set(taskId, task);
    }
    return task;
  }

  /**
   * 根据 eventId 获取 Event 包装对象。
   */
  getEvent(eventId: string): Event | undefined {
    const yMap = this.eventsMap.get(eventId);
    if (!yMap) return undefined;

    let event = this.eventCache.get(eventId);
    if (!event) {
      event = new Event(yMap, this);
      this.eventCache.set(eventId, event);
    }
    return event;
  }

  /**
   * 根据 textId 获取 Y.Text
   */
  getText(textId: string): Y.Text | undefined {
    return this.textsMap.get(textId);
  }

  /**
   * 清理缓存中已经不存在的实体（可选，用于 GC）
   */
  pruneCache(): void {
    for (const [id, _] of this.taskCache) {
      if (!this.tasksMap.has(id)) {
        this.taskCache.delete(id);
      }
    }
    for (const [id, _] of this.eventCache) {
      if (!this.eventsMap.has(id)) {
        this.eventCache.delete(id);
      }
    }
  }
}
```

### 响应式可迭代数组代理

```typescript
// reactive-array.ts
import * as Y from 'yjs';
import { createYArraySubscriber } from './reactive-ymap';

/**
 * ReactiveYArrayProxy<T> 包装一个 Y.Array<string>（存储 id 列表），
 * 通过 mapper 函数将每个 id 映射为实体对象 T。
 * 
 * 它实现了 Iterable<T>，并暴露 length 和 [index] 访问，
 * 使得在 Svelte 的 {#each} 中可以直接使用。
 * 
 * 关键：每次读取（迭代/访问 length/索引）都会调用 subscribe()，
 * 这样 Svelte 就能追踪到这个依赖。
 */
export class ReactiveYArrayProxy<T> implements Iterable<T> {
  private readonly yArray: Y.Array<string>;
  private readonly mapper: (id: string) => T | undefined;
  private readonly subscribe: () => void;

  constructor(yArray: Y.Array<string>, mapper: (id: string) => T | undefined) {
    this.yArray = yArray;
    this.mapper = mapper;
    this.subscribe = createYArraySubscriber(yArray);
  }

  get length(): number {
    this.subscribe(); // 触发 Svelte 依赖追踪
    return this.yArray.length;
  }

  /**
   * 获取第 i 个元素对应的实体
   */
  at(index: number): T | undefined {
    this.subscribe();
    const id = this.yArray.get(index);
    if (id === undefined) return undefined;
    return this.mapper(id);
  }

  /**
   * 返回所有 id 的原始数组快照
   */
  toIds(): string[] {
    this.subscribe();
    return this.yArray.toArray();
  }

  /**
   * 返回所有映射后的实体数组快照。
   * 在 {#each} 中使用时，Svelte 5 会调用 [Symbol.iterator]。
   */
  toArray(): T[] {
    this.subscribe();
    const result: T[] = [];
    for (let i = 0; i < this.yArray.length; i++) {
      const id = this.yArray.get(i);
      const entity = this.mapper(id);
      if (entity !== undefined) {
        result.push(entity);
      }
    }
    return result;
  }

  [Symbol.iterator](): Iterator<T> {
    // 调用 toArray() 内部会触发 subscribe
    const arr = this.toArray();
    let i = 0;
    return {
      next(): IteratorResult<T> {
        if (i < arr.length) {
          return { value: arr[i++], done: false };
        }
        return { value: undefined as any, done: true };
      }
    };
  }
}
```

```typescript
 * 3. children/parents/events 返回 ReactiveYArrayProxy，支持 {#each} 迭代
 * 4. 关联字段的 ReactiveYArrayProxy 懒初始化并缓存，避免每次 getter 都新建
 *

 *   const task = store.getTask('task-1');
```

### Event 类

```typescript
// event.svelte.ts
import * as Y from 'yjs';
import type { Store } from './store';
import type { Task } from './task.svelte';
import { createYMapSubscriber } from './reactive-ymap';

/**
 * Event 是对 ydoc.getMap('events') 中某个 Y.Map 的响应式包装。
 *
 * 字段比 Task 简单得多，只有标量字段和一个指向 Task 的反向引用。
 *
 * 使用示例：
 *   const event = store.getEvent('event-1');
 *   event.start       // number 时间戳
 *   event.start = Date.now()  // 写入，触发协同同步+响应式更新
 *   event.task        // 关联的 Task 对象
 */
export class Event {
  readonly yMap: Y.Map<any>;
  private readonly store: Store;
  private readonly subscribe: () => void;

  constructor(yMap: Y.Map<any>, store: Store) {
    this.yMap = yMap;
    this.store = store;
    this.subscribe = createYMapSubscriber(yMap);
  }

  // ─── 标量字段 ───────────────────────────────────────

  get id(): string {
    this.subscribe();
    return this.yMap.get('id');
  }

  get taskId(): string {
    this.subscribe();
    return this.yMap.get('taskId');
  }

  set taskId(value: string) {
    this.yMap.set('taskId', value);
  }

  get start(): number {
    this.subscribe();
    return this.yMap.get('start');
  }

  set start(value: number) {
    this.yMap.set('start', value);
  }

  get end(): number {
    this.subscribe();
    return this.yMap.get('end');
  }

  set end(value: number) {
    this.yMap.set('end', value);
  }

  // ─── 关联字段 ───────────────────────────────────────

  /**
   * 反向引用：通过 taskId 获取关联的 Task 对象。
   * 每次访问都是实时从 store 缓存中取，无需额外缓存。
   */
  get task(): Task | undefined {
    this.subscribe(); // 追踪 taskId 的变化
    const taskId = this.yMap.get('taskId');
    return taskId ? this.store.getTask(taskId) : undefined;
  }

  // ─── 便捷方法 ───────────────────────────────────────

  /**
   * 获取事件时长（毫秒）
   */
  get duration(): number {
    this.subscribe();
    return this.end - this.start;
  }

  toJSON(): Record<string, any> {
    return {
      id: this.id,
      taskId: this.taskId,
      start: this.start,
      end: this.end,
    };
  }
}
```

### Store 完整版（补充创建方法）

```typescript
// store.ts (完整版)
import * as Y from 'yjs';
import { Task } from './task.svelte';
import { Event } from './event.svelte';
import { createYMapKeysSubscriber } from './reactive-ymap';

export class Store {
  readonly ydoc: Y.Doc;
  readonly tasksMap: Y.Map<Y.Map<any>>;
  readonly eventsMap: Y.Map<Y.Map<any>>;
  readonly textsMap: Y.Map<Y.Text>;

  private taskCache = new Map<string, Task>();
  private eventCache = new Map<string, Event>();

  // 用于在 Svelte 组件中响应式地遍历所有 task / event
  private readonly subscribeTasksMap: () => void;
  private readonly subscribeEventsMap: () => void;

  private nextTextId = 0;

  constructor(ydoc: Y.Doc) {
    this.ydoc = ydoc;
    this.tasksMap = ydoc.getMap('tasks') as Y.Map<Y.Map<any>>;
    this.eventsMap = ydoc.getMap('events') as Y.Map<Y.Map<any>>;
    this.textsMap = ydoc.getMap('texts') as Y.Map<Y.Text>;

    this.subscribeTasksMap = createYMapKeysSubscriber(this.tasksMap);
    this.subscribeEventsMap = createYMapKeysSubscriber(this.eventsMap);
  }

  // ─── 获取（带缓存） ─────────────────────────────────

  getTask(taskId: string): Task | undefined {
    const yMap = this.tasksMap.get(taskId);
    if (!yMap) {
      this.taskCache.delete(taskId);
      return undefined;
    }

    let task = this.taskCache.get(taskId);
    if (!task) {
      task = new Task(yMap, this);
      this.taskCache.set(taskId, task);
    }
    return task;
  }

  getEvent(eventId: string): Event | undefined {
    const yMap = this.eventsMap.get(eventId);
    if (!yMap) {
      this.eventCache.delete(eventId);
      return undefined;
    }

    let event = this.eventCache.get(eventId);
    if (!event) {
      event = new Event(yMap, this);
      this.eventCache.set(eventId, event);
    }
    return event;
  }

  getText(textId: string): Y.Text | undefined {
    return this.textsMap.get(textId);
  }

  // ─── 响应式遍历所有实体 ─────────────────────────────

  /**
   * 获取所有 Task，响应式。
   * 当 tasksMap 中有 key 增删时，Svelte 会重新读取。
   * 适用于：{#each store.allTasks as task (task.id)}
   */
  get allTasks(): Task[] {
    this.subscribeTasksMap();
    const result: Task[] = [];
    for (const [id] of this.tasksMap) {
      const task = this.getTask(id);
      if (task) result.push(task);
    }
    return result;
  }

  get allEvents(): Event[] {
    this.subscribeEventsMap();
    const result: Event[] = [];
    for (const [id] of this.eventsMap) {
      const event = this.getEvent(id);
      if (event) result.push(event);
    }
    return result;
  }

  // ─── 创建实体 ───────────────────────────────────────

  /**
   * 创建一个新的 Task，初始化所有必要的 yjs 结构。
   * 所有写操作包裹在 transact 中，确保原子性。
   */
  createTask(taskId: string): Task {
    const textId = this.generateTextId();

    this.ydoc.transact(() => {
      // 创建关联的 Y.Text
      const yText = new Y.Text();
      this.textsMap.set(textId, yText);

      // 创建 task 的 Y.Map
      const taskYMap = new Y.Map();
      taskYMap.set('id', taskId);
      taskYMap.set('textId', textId);
      taskYMap.set('noteDoc', new Y.Doc());
      taskYMap.set('children', new Y.Array());
      taskYMap.set('parents', new Y.Array());
      taskYMap.set('events', new Y.Array());

      this.tasksMap.set(taskId, taskYMap);
    });

    return this.getTask(taskId)!;
  }

  /**
   * 创建一个新的 Event 并自动关联到 Task。
   */
  createEvent(eventId: string, taskId: string, start: number, end: number): Event {
    this.ydoc.transact(() => {
      const eventYMap = new Y.Map();
      eventYMap.set('id', eventId);
      eventYMap.set('taskId', taskId);
      eventYMap.set('start', start);
      eventYMap.set('end', end);

      this.eventsMap.set(eventId, eventYMap);

      // 自动在 task 的 events 数组中添加引用
      const taskYMap = this.tasksMap.get(taskId);
      if (taskYMap) {
        const eventsArray = taskYMap.get('events') as Y.Array<string>;
        eventsArray.push([eventId]);
      }
    });

    return this.getEvent(eventId)!;
  }

  // ─── 删除实体 ───────────────────────────────────────

  deleteTask(taskId: string): void {
    const task = this.getTask(taskId);
    if (!task) return;

    this.ydoc.transact(() => {
      // 清理关联的 text
      const textId = task.yMap.get('textId');
      if (textId) this.textsMap.delete(textId);

      // 从所有 parent 的 children 中移除自己
      const parents = (task.yMap.get('parents') as Y.Array<string>).toArray();
      for (const parentId of parents) {
        const parentYMap = this.tasksMap.get(parentId);
        if (parentYMap) {
          const childrenArr = parentYMap.get('children') as Y.Array<string>;
          const idx = childrenArr.toArray().indexOf(taskId);
          if (idx !== -1) childrenArr.delete(idx, 1);
        }
      }

      // 从所有 child 的 parents 中移除自己
      const children = (task.yMap.get('children') as Y.Array<string>).toArray();
      for (const childId of children) {
        const childYMap = this.tasksMap.get(childId);
        if (childYMap) {
          const parentsArr = childYMap.get('parents') as Y.Array<string>;
          const idx = parentsArr.toArray().indexOf(taskId);
          if (idx !== -1) parentsArr.delete(idx, 1);
        }
      }

      // 删除关联的 events
      const eventIds = (task.yMap.get('events') as Y.Array<string>).toArray();
      for (const eid of eventIds) {
        this.eventsMap.delete(eid);
        this.eventCache.delete(eid);
      }

      this.tasksMap.delete(taskId);
      this.taskCache.delete(taskId);
    });
  }

  deleteEvent(eventId: string): void {
    const event = this.getEvent(eventId);
    if (!event) return;

    this.ydoc.transact(() => {
      // 从关联 task 的 events 数组中移除
      const taskYMap = this.tasksMap.get(event.taskId);
      if (taskYMap) {
        const eventsArr = taskYMap.get('events') as Y.Array<string>;
        const idx = eventsArr.toArray().indexOf(eventId);
        if (idx !== -1) eventsArr.delete(idx, 1);
      }

      this.eventsMap.delete(eventId);
      this.eventCache.delete(eventId);
    });
  }

  // ─── 工具 ──────────────────────────────────────────

  private generateTextId(): string {
    return `text-${++this.nextTextId}-${Date.now()}`;
  }

  pruneCache(): void {
    for (const [id] of this.taskCache) {
      if (!this.tasksMap.has(id)) this.taskCache.delete(id);
    }
    for (const [id] of this.eventCache) {
      if (!this.eventsMap.has(id)) this.eventCache.delete(id);
    }
  }
}
```

### 完整的初始化入口（带协同连接）

```typescript
// init.ts
import * as Y from 'yjs';
import { WebsocketProvider } from 'y-websocket';
import { Store } from './store';

/**
 * 创建并初始化 Store 实例。
 * 可选连接 WebSocket provider 实现多端协同。
 */
export function createStore(options?: {
  roomName?: string;
  wsUrl?: string;
}): { store: Store; destroy: () => void } {
  const ydoc = new Y.Doc();
  const store = new Store(ydoc);

  let provider: WebsocketProvider | undefined;

  if (options?.roomName && options?.wsUrl) {
    provider = new WebsocketProvider(options.wsUrl, options.roomName, ydoc);
  }

  return {
    store,
    destroy() {
      provider?.destroy();
      ydoc.destroy();
    },
  };
}
```

## 设计决策总结

| 决策 | 原因 |
|------|------|
| **实例缓存（Store 中的 taskCache/eventCache）** | 同一个 id 始终返回同一个 JS 对象引用，对 Svelte 的 keyed `{#each}` 友好，避免重复创建 `createSubscriber` |
| **每个 Y.Map/Y.Array 独立 `createSubscriber`** | 细粒度响应式：修改某个 event 的 start 不会触发其他 event 或 task 的重渲染 |
| **getter 中调用 `subscribe()`** | 只有实际被读取的字段才注册依赖，未读取的不会造成多余更新 |
| **ReactiveYArrayProxy 实现 `Iterable`** | 直接用于 `{#each}`，API 简洁；内部 `toArray()` 时做一次快照，迭代期间稳定 |
| **懒初始化 `_children` / `_events` 等代理** | 避免构造 Task 时就创建所有代理，只有首次访问时才初始化 |
| **setter 直接操作 yMap，不手动调用 update** | yjs 的 observe 机制会自动触发 `createSubscriber` 的 update，形成闭环 |
| **`transact` 包裹多步写操作** | 保证原子性：多个 yMap/yArray 的写操作合并为一次 yjs 事务，只触发一次 observe 批量通知 |
