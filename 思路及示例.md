# 提取 Yjs 文档的纯文本摘要

直接从 Yjs 的 XmlFragment 中提取文本。

## 核心工具函数

```typescript
import * as Y from 'yjs'

/**
 * 从 Yjs update 中提取文档摘要文本
 * @param update - Yjs 文档的二进制 update
 * @param fieldName - TipTap 使用的 fragment 名称，默认 'default'
 * @param maxLength - 最大截取长度
 */
export function extractSummary(
  ydoc: Y.Doc,
  fieldName = 'default',
  maxLength = 200
): string {

  const xmlFragment = ydoc.getXmlFragment(fieldName)
  const text = extractTextFromFragment(xmlFragment, maxLength)

  return text.slice(0, maxLength)
}

function extractTextFromFragment(fragment: Y.XmlFragment, maxLength: number): string {
  const parts: string[] = []
  let totalLength = 0

  for (const child of fragment.toArray()) {
    if (totalLength >= maxLength) break

    const text = extractTextFromNode(child)
    if (text) {
      parts.push(text)
      totalLength += text.length
    }
  }

  return parts.join(' ')
}

function extractTextFromNode(node: Y.XmlElement | Y.XmlText | any): string {
  // 文本节点
  if (node instanceof Y.XmlText) {
    return node.toString()
  }

  // 元素节点
  if (node instanceof Y.XmlElement) {
    const nodeName = node.nodeName

    // 图片 → 替换为占位符
    if (nodeName === 'image') {
      return '[图片]'
    }

    // 水平线
    if (nodeName === 'horizontalRule') {
      return ''
    }

    // 代码块 → 简化展示
    if (nodeName === 'codeBlock') {
      return '[代码]'
    }

    // 其他节点：递归提取子节点文本
    const children = node.toArray()
    const texts = children.map((child: any) => extractTextFromNode(child)).filter(Boolean)
    return texts.join('')
  }

  return ''
}
```

## 使用

```typescript
const summary = extractSummary(yDoc, 'default', 100)

console.log(summary) // "这是文档的第一行内容 [图片] 后面还有一些文字..."
```

## 如果你只要第一行

```typescript
/**
 * 只提取第一个有内容的块级节点的文本
 */
export function extractFirstLine(
  update: Uint8Array,
  fieldName = 'default',
  maxLength = 100
): string {
  const ydoc = new Y.Doc()
  Y.applyUpdate(ydoc, update)

  const xmlFragment = ydoc.getXmlFragment(fieldName)

  for (const child of xmlFragment.toArray()) {
    const text = extractTextFromNode(child)
    if (text.trim()) {
      return text.trim().slice(0, maxLength)
    }
    // 如果第一个块是图片等，也返回
    if (child instanceof Y.XmlElement && child.nodeName === 'image') {
      return '[图片]'
    }
  }

  return ''
}
```
